SELECT 
    n.TN001 AS 單別, 
    n.TN002 AS 單號, 
    n.TN003 AS 單據日期, 
    n.TN021 AS 確認碼, 
    CASE n.TN021 
        WHEN 'Y' THEN '已核單'
        ELSE n.TN021 
    END AS 單據狀態,
    n.TN004 AS 客戶編號, 
    n.TN005 AS 業務員編號, 
    o.TO003 AS 項次, 
    o.TO004 AS 料件編號, 
    o.TO005 AS 品名規格, 
    o.TO009 AS 數量, 
    o.TO016 AS 預交日
FROM COPTN n
JOIN COPTO o ON n.TN001 = o.TO001 AND n.TN002 = o.TO002
WHERE n.TN001 = '2B01'
  AND n.TN021 = 'Y'
  AND (
      n.TN002 BETWEEN '20110102001' AND '20110102296' OR
      n.TN002 BETWEEN '20110103001' AND '20110103296' OR
      n.TN002 BETWEEN '20110104001' AND '20110104296' OR
      n.TN002 BETWEEN '20110106001' AND '20110106296' OR
      n.TN002 BETWEEN '20110107001' AND '20110107296' OR
      n.TN002 BETWEEN '20110108001' AND '20110108296' OR
      n.TN002 BETWEEN '20110112001' AND '20110112296' OR
      n.TN002 BETWEEN '20110114001' AND '20110114296' OR
      n.TN002 BETWEEN '20110116001' AND '20110116296' OR
      n.TN002 BETWEEN '20110118001' AND '20110118296' OR
      n.TN002 BETWEEN '20110119001' AND '20110119296'
  )
